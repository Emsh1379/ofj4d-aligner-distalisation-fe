# FEBioAlignerContact plug-in: contact "aligner-contact". See FEAlignerContact.cpp.
cmake_minimum_required(VERSION 3.18)
project(FEBioAlignerContact LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
find_package(OpenMP REQUIRED)
set(FEBIO_ROOT "" CACHE PATH "FEBio source root (contains FECore/)")
set(FEBIO_LIB_DIR "" CACHE PATH "FEBio build library directory")
add_library(FEBioAlignerContact SHARED FEAlignerContact.cpp)
target_compile_definitions(FEBioAlignerContact PRIVATE LINUX)
target_include_directories(FEBioAlignerContact PRIVATE ${FEBIO_ROOT} ${FEBIO_ROOT}/FEBioMech)
target_link_directories(FEBioAlignerContact PRIVATE ${FEBIO_LIB_DIR})
target_link_libraries(FEBioAlignerContact PRIVATE febiomech fecore OpenMP::OpenMP_CXX)
